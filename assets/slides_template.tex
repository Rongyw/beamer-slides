\section{Design}

\begin{frame}{Identification: within a unit-year, months with an unusually large shock}
\begin{itemize}
\item \textbf{Shock:} $Z_{bt}$ = predicted deviation from the calendar-month norm\\
  {\small one line on how it is built and what it is scaled by}
\item \textbf{Estimating equation:} $y_{it} = \beta X_{it} + W'\gamma + \alpha_{i\times y} + \alpha_{i\times m} + \delta_t +
  \varepsilon_{it}$, $X$ instrumented by $Z$
\item \textbf{Sample:} source, units, years
\item \textbf{Exclusion checks:} $Z$ does not predict demand, weather controls or placebo outcomes; leads $\approx 0$
\item \textbf{Inference:} clustered by region, plus wild-cluster bootstrap \citeg{Roodman et al., 2019}
\end{itemize}
\takeaway{the shock is the experiment; the fixed effects remove levels, seasonality and common monthly shocks.}
\end{frame}

\section{Robustness}

% Pattern 8: problem / solution
\begin{frame}{Two-way fixed effects flip the sign under staggered timing}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\small
\textbf{The problem}
\begin{itemize}
\item Treatment arrives at different dates and its effect changes over time
\item Two-way fixed effects use already-treated units as controls; weights can be negative \citeg{Goodman-Bacon, 2021}
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\small
\textbf{Why the chosen estimator fits}
\begin{itemize}
\item Each cohort compared only with never- or not-yet-treated units \citeg{Callaway \& Sant'Anna, 2021}
\item Other estimators reported as checks
\end{itemize}
\end{column}
\end{columns}
\takeaway{clean comparisons and explicit aggregation are what a staggered, dynamic policy requires.}
\end{frame}
